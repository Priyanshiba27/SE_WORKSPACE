1.Create a file called playlist.txt and write the names of your top 3 favorite songs from Spotify into it using write mode (w).

#include <stdio.h>

int main()
{
    FILE *file;

    file = fopen("playlist.txt", "w");

    fprintf(file, "Perfect\n");
    fprintf(file, "Love Story\n");
    fprintf(file, "Shape of You\n");

    fclose(file);

    printf("3 songs written to playlist.txt\n");

    return 0;
}

4.Write a program that reads all song names from playlist.txt and prints only those that contain the word 'love' (case-insensitive).

#include <stdio.h>
#include <string.h>

int main()
{
    FILE *file;
    char song[100];
    char lower[100];

    file = fopen("playlist.txt", "r");

    while(fgets(song, 100, file) != NULL)
    {
        strcpy(lower, song);


        for(int i = 0; lower[i] != '\0'; i++)
        {
            if(lower[i] >= 'A' && lower[i] <= 'Z')
            {
                lower[i] = lower[i] + 32;
            }
        }

       
        if(strstr(lower, "love") != NULL)
        {
            printf("%s", song);
        }
    }

    fclose(file);

    return 0;
}
