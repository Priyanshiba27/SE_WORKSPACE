
#include <iostream>
#include <fstream>
using namespace std;

class Content
{
public:
    string title;
    string platform;
    int views;
    string status;
};

int main()
{
    Content c;
    int choice, number;

    while (true)
    {
        cout << "\n1. Add Content";
        cout << "\n2. Display Content";
        cout << "\n3. Update Status";
        cout << "\n4. Delete Content";
        cout << "\n5. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;
        cin.ignore();

        // Add
        if (choice == 1)
        {
            cout << "Enter title: ";
            getline(cin, c.title);

            cout << "Enter platform: ";
            getline(cin, c.platform);

            cout << "Enter views: ";
            cin >> c.views;
            cin.ignore();

            cout << "Enter status: ";
            getline(cin, c.status);

            ofstream file("content_list.txt", ios::app);

            file << c.title << endl;
            file << c.platform << endl;
            file << c.views << endl;
            file << c.status << endl;

            file.close();
        }

        // Display
        else if (choice == 2)
        {
            ifstream file("content_list.txt");

            int i = 1;

            while (getline(file, c.title))
            {
                getline(file, c.platform);
                getline(file, c.status);

                cout << i << ". "
                     << c.title << " - "
                     << c.platform << endl;

                i++;
            }

            file.close();
        }

        // Update
        else if (choice == 3)
        {
            cout << "Enter content number: ";
            cin >> number;
            cin.ignore();

            ifstream file("content_list.txt");
            ofstream temp("temp.txt");

            int i = 1;

            while (getline(file, c.title))
            {
                getline(file, c.platform);
                file >> c.views;
                file.ignore();
                getline(file, c.status);

                if (i == number)
                {
                    cout << "Enter new status: ";
                    getline(cin, c.status);
                }

                temp << c.title << endl;
                temp << c.platform << endl;
                temp << c.views << endl;
                temp << c.status << endl;

                i++;
            }

            file.close();
            temp.close();

            remove("content_list.txt");
            rename("temp.txt", "content_list.txt");
        }

        // Delete
        else if (choice == 4)
        {
            cout << "Enter content number: ";
            cin >> number;

            ifstream file("content_list.txt");
            ofstream temp("temp.txt");

            int i = 1;

            while (getline(file, c.title))
            {
                getline(file, c.platform);
                file >> c.views;
                file.ignore();
                getline(file, c.status);

                if (i != number)
                {
                    temp << c.title << endl;
                    temp << c.platform << endl;
                    temp << c.views << endl;
                    temp << c.status << endl;
                }

                i++;
            }

            file.close();
            temp.close();

            remove("content_list.txt");
            rename("temp.txt", "content_list.txt");

            cout << "Content deleted." << endl;
        }

        // Exit
        else if (choice == 5)
        {
            break;
        }
    }

    return 0;
}
