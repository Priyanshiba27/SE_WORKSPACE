4. Extend your Playlist class to include a member function addSong(songTitle) that adds the song title to an array property called songs. Demonstrate by adding three song titles and displaying the updated songs list.<br><br><em><strong>Hint:</strong> Initialize songs as an empty array inside the constructor.</em>

#include <iostream>
#include <string>
using namespace std;

class Playlist
{
public:
    string songs[10];
    int songCount=0;


     
      void addSong(string songTitle)
    {
        songs[songCount] = songTitle;
        songCount++;
    }

   
    void displaySongs()
    {
        cout << "Songs in Playlist:" << endl;
        for (int i = 0; i < songCount; i++)
        {
            cout << songs[i] << endl;
        }
    }
};

int main()
{
    Playlist myPlaylist;

    myPlaylist.addSong("Tum Hi Ho");
    myPlaylist.addSong("Kesariya");
    myPlaylist.addSong("Apna Bana Le");

    myPlaylist.displaySongs();

    return 0;
}

5.Refactor your FoodOrder class so that the constructor takes an object as a parameter (with keys orderId, restaurantName, isDelivered) instead of separate arguments. Update your instantiation code to use this new constructor signature.

#include <iostream>
using namespace std;

class FoodOrder
{
public:
    int orderId;
    string restaurantName;
    bool isDelivered;

    FoodOrder(int id, string name, bool delivered)
    {
        orderId = id;
        restaurantName = name;
        isDelivered = delivered;
    }

    void markDelivered()
    {
        isDelivered = true;
        cout << "Order delivered successfully!" << endl;
    }

    void display()
    {
        cout << "Order ID: " << orderId << endl;
        cout << "Restaurant: " << restaurantName << endl;

        if (isDelivered)
            cout << "Status: Delivered" << endl;
        else
            cout << "Status: Not Delivered" << endl;
    }
};

int main()
{
    FoodOrder order(101, "Zomato", false);

    order.display();

    order.markDelivered();

    order.display();

    return 0;
}
